createEmbeddingChunksTask = async (id: string) => {
  const { userId } = getCtx();
  if (!userId) throw new Error('createEmbeddingChunksTask requires userId');
  return apiPost('/api/lobechat-kb/batch-embed-file', { file_id: id, user_id: userId });
};
